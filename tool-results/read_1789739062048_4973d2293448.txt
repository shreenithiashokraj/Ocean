   123→   123→const COMPASS_POINTS = [
   124→   124→  "N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE",
   125→   125→  "S", "SSW", "SW", "WSW", "W", "WNW", "NW", "NNW",

   127→   127→
   128→   128→function degToCompass(deg: number | null | undefined): string | null {
   129→   129→  if (deg == null || Number.isNaN(deg)) return null;
   130→   130→  return COMPASS_POINTS[Math.round(deg / 22.5) % 16];
   131→   131→}
   132→   132→
   133→   133→function fmtNum(n: number | null | undefined, digits = 1): string | null {
   134→   134→  if (n == null || Number.isNaN(n)) return null;
   135→   135→  return n.toFixed(digits);
   136→   136→}
   137→   137→
   138→   138→/** Extract "HH:MM" from an open-meteo local-time string like "2024-01-01T06:30". */
   139→   139→function formatHHMM(iso: string | null | undefined): string | null {
   140→   140→  if (!iso) return null;
   141→   141→  if (iso.length >= 16) return iso.slice(11, 16);
   142→   142→  return iso;
   143→   143→}
   144→   144→
   145→   145→function formatUpdatedAt(iso: string | null): string {
   146→   146→  if (!iso) return "N/A";
   147→   147→  const d = new Date(iso);
   148→   148→  if (Number.isNaN(d.getTime())) return iso;
   149→   149→  return d.toLocaleString();
   150→   150→}
   151→   151→
   152→   152→type Safety = "safe" | "caution" | "dangerous" | "unknown";
   153→   153→
   154→   154→/** Derive safety status from wind speed (km/h) and wave height (m). */
   155→   155→function deriveStatus(data: SeaConditionData | null): Safety {
   156→   156→  if (!data) return "unknown";
   157→   157→  const wind = data.windSpeed;
   158→   158→  const wave = data.waveHeight;
   159→   159→  if (wind == null && wave == null) return "unknown";
   160→   160→  const w = wind ?? 0;
   161→   161→  const h = wave ?? 0;
   162→   162→  if (w < 25 && h < 1.5) return "safe";
   163→   163→  if (w < 40 && h < 2.5) return "caution";
   164→   164→  return "dangerous";
   165→   165→}

   214→   214→      setIsDemo(res.isDemo);

   216→   216→      setSource(res.source);

   220→   220→        ? err.message
   221→   221→        : err instanceof Error
   222→   222→          ? err.message
   223→   223→          : "Failed to fetch sea conditions";
   224→   224→      setError(msg);
   225→   225→      if (!silent) toast.error("Failed to load sea conditions.");

   239→   239→      (pos) => {
   240→   240→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   241→   241→        setAccuracy(pos.coords.accuracy ?? null);
   242→   242→      },
   243→   243→      () => {
   244→   244→        toast.info("Location access denied — showing Chennai by default.");
   245→   245→        setCoords(CHENNAI);
   246→   246→      },
   247→   247→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 60000 },
   248→   248→    );
   249→   249→  }, []);
   250→   250→
   251→   251→  // ── Fetch whenever coords change ─────────────────────────────────

   262→   262→      void fetchData(coords.lat, coords.lng, true);

   271→   271→      setSearchResults([]);

   301→   301→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 },

   372→   372→              <SelectTrigger
