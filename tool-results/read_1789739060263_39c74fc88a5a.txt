   123→const COMPASS_POINTS = [
   124→  "N", "NNE", "NE", "ENE", "E", "ESE", "SE", "SSE",
   125→  "S", "SSW", "SW", "WSW", "W", "WNW", "NW", "NNW",
   126→];
   127→
   128→function degToCompass(deg: number | null | undefined): string | null {
   129→  if (deg == null || Number.isNaN(deg)) return null;
   130→  return COMPASS_POINTS[Math.round(deg / 22.5) % 16];
   131→}
   132→
   133→function fmtNum(n: number | null | undefined, digits = 1): string | null {
   134→  if (n == null || Number.isNaN(n)) return null;
   135→  return n.toFixed(digits);
   136→}
   137→
   138→/** Extract "HH:MM" from an open-meteo local-time string like "2024-01-01T06:30". */
   139→function formatHHMM(iso: string | null | undefined): string | null {
   140→  if (!iso) return null;
   141→  if (iso.length >= 16) return iso.slice(11, 16);
   142→  return iso;
   143→}
   144→
   145→function formatUpdatedAt(iso: string | null): string {
   146→  if (!iso) return "N/A";
   147→  const d = new Date(iso);
   148→  if (Number.isNaN(d.getTime())) return iso;
   149→  return d.toLocaleString();
   150→}
   151→
   152→type Safety = "safe" | "caution" | "dangerous" | "unknown";
   153→
   154→/** Derive safety status from wind speed (km/h) and wave height (m). */
   155→function deriveStatus(data: SeaConditionData | null): Safety {
   156→  if (!data) return "unknown";
   157→  const wind = data.windSpeed;
   158→  const wave = data.waveHeight;
   159→  if (wind == null && wave == null) return "unknown";
   160→  const w = wind ?? 0;
   161→  const h = wave ?? 0;
   162→  if (w < 25 && h < 1.5) return "safe";
   163→  if (w < 40 && h < 2.5) return "caution";
   164→  return "dangerous";
   165→}

   167→// WMO weather code → lucide icon mapping (Open-Meteo uses WMO codes).
   168→const WEATHER_ICONS: Record<number, typeof Sun> = {
   169→  0: Sun,
   170→  1: CloudSun,
   171→  2: CloudSun,
   172→  3: Cloud,
   173→  45: CloudFog,
   174→  48: CloudFog,
   175→  51: CloudDrizzle, 53: CloudDrizzle, 55: CloudDrizzle, 56: CloudDrizzle, 57: CloudDrizzle,
   176→  61: CloudRain, 63: CloudRain, 65: CloudRain, 66: CloudRain, 67: CloudRain,
   177→  71: Snowflake, 73: Snowflake, 75: Snowflake, 77: Snowflake,
   178→  80: CloudRain, 81: CloudRain, 82: CloudRain,
   179→  85: Snowflake, 86: Snowflake,
   180→  95: CloudLightning, 96: CloudLightning, 99: CloudLightning,
   181→};
   182→
   183→function weatherIconFor(code: number | null | undefined) {
   184→  if (code == null) return Cloud;
   185→  return WEATHER_ICONS[code] ?? Cloud;
   186→}

   190→export function SeaConditionsView() {
   191→  const [coords, setCoords] = useState<{ lat: number; lng: number } | null>(null);
   192→  const [accuracy, setAccuracy] = useState<number | null>(null);
   193→  const [label, setLabel] = useState<string>("");
   194→  const [data, setData] = useState<SeaConditionData | null>(null);
   195→  const [isDemo, setIsDemo] = useState<boolean>(false);
   196→  const [recordedAt, setRecordedAt] = useState<string | null>(null);
   197→  const [source, setSource] = useState<string>("");
   198→  const [loading, setLoading] = useState<boolean>(false);
   199→  const [error, setError] = useState<string | null>(null);
   200→  const [refreshMinutes, setRefreshMinutes] = useState<RefreshMinutes>("off");
   201→  const [searchQuery, setSearchQuery] = useState<string>("");
   202→  const [searching, setSearching] = useState<boolean>(false);
   203→  const [searchResults, setSearchResults] = useState<PlaceResult[]>([]);
   204→  const [showResults, setShowResults] = useState<boolean>(false);
   205→  const searchContainerRef = useRef<HTMLDivElement>(null);
   206→
   207→  // ── Data fetcher ──────────────────────────────────────────────────
   208→  const fetchData = useCallback(async (lat: number, lng: number, silent = false) => {
   209→    if (!silent) setLoading(true);
   210→    setError(null);
   211→    try {
   212→      const res = (await api.getSeaConditions(lat, lng)) as unknown as SeaConditionsResponse;
   213→      setData(res.data);
   214→      setIsDemo(res.isDemo);
   215→      setRecordedAt(res.recordedAt);
   216→      setSource(res.source);
   217→      setLabel(res.label);
   218→    } catch (err) {
   219→      const msg = err instanceof ApiError
   220→        ? err.message
   221→        : err instanceof Error
   222→          ? err.message
   223→          : "Failed to fetch sea conditions";
   224→      setError(msg);
   225→      if (!silent) toast.error("Failed to load sea conditions.");
   226→    } finally {
   227→      if (!silent) setLoading(false);
   228→    }
   229→  }, []);
   230→
   231→  // ── On mount: geolocate or fall back to Chennai ───────────────────
   232→  useEffect(() => {
   233→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   234→      toast.info("Geolocation is unavailable on this device — showing Chennai by default.");
   235→      setCoords(CHENNAI);
   236→      return;
   237→    }
   238→    navigator.geolocation.getCurrentPosition(
   239→      (pos) => {
   240→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   241→        setAccuracy(pos.coords.accuracy ?? null);
   242→      },
   243→      () => {
   244→        toast.info("Location access denied — showing Chennai by default.");
   245→        setCoords(CHENNAI);
   246→      },
   247→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 60000 },
   248→    );
   249→  }, []);
   250→
   251→  // ── Fetch whenever coords change ─────────────────────────────────
   252→  useEffect(() => {
   253→    if (!coords) return;
   254→    void fetchData(coords.lat, coords.lng);
   255→  }, [coords, fetchData]);
   256→
   257→  // ── Auto-refresh interval ─────────────────────────────────────────
   258→  useEffect(() => {
   259→    if (!coords || refreshMinutes === "off") return;
   260→    const ms = Number(refreshMinutes) * 60 * 1000;
   261→    const t = setInterval(() => {
   262→      void fetchData(coords.lat, coords.lng, true);
   263→    }, ms);
   264→    return () => clearInterval(t);
   265→  }, [coords, refreshMinutes, fetchData]);
   266→
   267→  // ── Clear search dropdown when query is cleared ─────────────────
   268→  useEffect(() => {
   269→    if (searchQuery === "") {
   270→      setShowResults(false);
   271→      setSearchResults([]);
   272→    }
   273→  }, [searchQuery]);
   274→
   275→  // ── Click-outside to dismiss search dropdown ─────────────────────
   276→  useEffect(() => {
   277→    if (!showResults) return;
   278→    function onDocClick(e: MouseEvent) {
   279→      if (searchContainerRef.current && !searchContainerRef.current.contains(e.target as Node)) {
   280→        setShowResults(false);
   281→      }
   282→    }
   283→    document.addEventListener("mousedown", onDocClick);
   284→    return () => document.removeEventListener("mousedown", onDocClick);
   285→  }, [showResults]);
   286→
   287→  // ── "Use my location" button ──────────────────────────────────────
   288→  const useMyLocation = useCallback(() => {
   289→    if (typeof navigator === "undefined" || !navigator.geolocation) {
   290→      toast.error("Geolocation is not supported on this device.");
   291→      return;
   292→    }
   293→    toast.info("Requesting your location…");
   294→    navigator.geolocation.getCurrentPosition(
   295→      (pos) => {
   296→        setCoords({ lat: pos.coords.latitude, lng: pos.coords.longitude });
   297→        setAccuracy(pos.coords.accuracy ?? null);
   298→        toast.success("Using your current location.");
   299→      },
   300→      () => toast.error("Could not access your location."),
   301→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 },
   302→    );
   303→  }, []);
   304→
   305→  // ── Manual place search ───────────────────────────────────────────
   306→  const runSearch = useCallback(async () => {
   307→    const q = searchQuery.trim();
   308→    if (!q) return;
   309→    setSearching(true);
   310→    setShowResults(true);
   311→    try {
   312→      const res = (await api.searchPlaces(q)) as unknown as PlacesResponse;
   313→      setSearchResults(res.results ?? []);
   314→      if ((res.results ?? []).length === 0) {
   315→        toast.info("No matching places found.");
   316→      }
   317→    } catch (err) {
   318→      toast.error(err instanceof Error ? err.message : "Place search failed.");
   319→      setSearchResults([]);
   320→    } finally {
   321→      setSearching(false);
   322→    }
   323→  }, [searchQuery]);
   324→
   325→  const pickPlace = useCallback((p: PlaceResult) => {
   326→    setCoords({ lat: p.lat, lng: p.lng });
   327→    setAccuracy(null);
   328→    setShowResults(false);
   329→    setSearchQuery("");
   330→    setSearchResults([]);
   331→    toast.success(`Location set to ${p.name}.`);
   332→  }, []);
   333→
   334→  const status = deriveStatus(data);
   335→
   336→  // ── Build chart dataset from hourly arrays ───────────────────────
   337→  const chartData = useMemo(() => {
   338→    const hourly = data?.hourly;
   339→    if (!hourly || !hourly.time || hourly.time.length === 0) return [];
   340→    return hourly.time.map((t, i) => ({
   341→      time: t.length >= 16 ? t.slice(11, 16) : t,
   342→      waveHeight: hourly.waveHeight?.[i] ?? null,
   343→      windSpeed: hourly.windSpeed?.[i] ?? null,
   344→    }));
   345→  }, [data]);
   346→
   347→  return (
   348→    <div className="mx-auto max-w-7xl px-4 py-6 sm:px-6">
   349→      {/* ── Page header ─────────────────────────────────────────────── */}
   350→      <header className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
   351→        <div>
   352→          <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   353→            Real-Time Sea Conditions
   354→          </h1>
   355→          <p className="mt-1 text-sm text-muted-foreground">
   356→            Live weather & marine data for safer fishing decisions. Source:{" "}
   357→            <span className="font-medium text-foreground">{source || "open-meteo"}</span>
   358→            {recordedAt && (
   359→              <span className="text-muted-foreground">
   360→                {" · "}Updated {formatUpdatedAt(recordedAt)}
   361→              </span>
   362→            )}
   363→          </p>
   364→        </div>
   365→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   366→          <label className="flex items-center gap-2" htmlFor="refresh-select">
   367→            <span className="text-xs font-medium text-muted-foreground">Auto-refresh</span>
   368→            <Select
   369→              value={refreshMinutes}
   370→              onValueChange={(v) => setRefreshMinutes(v as RefreshMinutes)}
   371→            >
   372→              <SelectTrigger
   373→                id="refresh-select"
   374→                className="w-[160px]"
   375→                aria-label="Auto-refresh interval"
   376→              >
   377→                <SelectValue />
   378→              </SelectTrigger>
   379→              <SelectContent>
   380→                {REFRESH_OPTIONS.map((o) => (
   381→                  <SelectItem key={o.value} value={o.value}>
   382→                    {o.label}
   383→                  </SelectItem>
   384→                ))}
   385→              </SelectContent>
   386→            </Select>
   387→          </label>
   388→          <Button
   389→            variant="outline"
   390→            size="sm"
   391→            onClick={useMyLocation}
   392→            aria-label="Use my current location"
   393→          >
   394→            <LocateFixed className="size-4" />
   395→            My location
   396→          </Button>
   397→        </div>
   398→      </header>
   399→
   400→      {/* ── Manual location search ──────────────────────────────────── */}
   401→      <div className="relative mt-4" ref={searchContainerRef}>
   402→        <div className="flex gap-2">
   403→          <div className="relative flex-1">
   404→            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   405→            <Input
   406→              type="search"
   407→              value={searchQuery}
   408→              onChange={(e) => setSearchQuery(e.target.value)}
   409→              onKeyDown={(e) => {
   410→                if (e.key === "Enter") {
   411→                  e.preventDefault();
   412→                  void runSearch();
   413→                }
   414→              }}
   415→              placeholder="Search a coastal city or place…"
   416→              className="pl-9"
   417→              aria-label="Search for a place by name"
   418→              aria-expanded={showResults}
   419→              aria-controls="place-results"
   420→            />
   421→          </div>
   422→          <Button onClick={() => void runSearch()} disabled={searching} size="sm">
   423→            {searching ? (
   424→              <Loader2 className="size-4 animate-spin" />
   425→            ) : (
   426→              <Search className="size-4" />
   427→            )}
   428→            Search
   429→          </Button>
   430→        </div>
   431→        {showResults && searchResults.length > 0 && (
   432→          <div
   433→            id="place-results"
   434→            role="listbox"
   435→            className="absolute z-50 mt-1 w-full overflow-hidden rounded-md border bg-popover shadow-lg"
   436→          >
   437→            <ul className="max-h-72 overflow-y-auto scroll-ocean py-1">
   438→              {searchResults.map((r, i) => (
   439→                <li key={`${r.name}-${r.lat}-${i}`} role="option" aria-selected={false}>
   440→                  <button
   441→                    type="button"
   442→                    onClick={() => pickPlace(r)}
   443→                    className="flex w-full flex-col items-start gap-0.5 px-3 py-2 text-left text-sm hover:bg-accent focus:bg-accent focus:outline-none"
   444→                  >
   445→                    <span className="font-medium">{r.name}</span>
   446→                    <span className="text-xs text-muted-foreground">
   447→                      {[r.region, r.country].filter(Boolean).join(", ") || "—"}
   448→                    </span>
   449→                  </button>
   450→                </li>
   451→              ))}
   452→            </ul>
   453→          </div>
   454→        )}
   455→      </div>
   456→
   457→      {/* ── Status banner ───────────────────────────────────────────── */}
   458→      <section className="mt-4" aria-live="polite">
   459→        <StatusBanner status={status} isDemo={isDemo} data={data} />
   460→      </section>
   461→
   462→      {/* ── Headline metrics strip (StatCard) ─────────────────────── */}
   463→      {data && !error && (
   464→        <div className="mt-4 grid grid-cols-2 gap-3 sm:grid-cols-4">
   465→          <StatCard
   466→            label="Wind Speed"
   467→            value={fmtNum(data.windSpeed, 0)}
   468→            unit="km/h"
   469→            icon={Wind}
   470→            accent="cyan"
   471→            sub={degToCompass(data.windDirection) ? `from ${degToCompass(data.windDirection)}` : undefined}
   472→          />
   473→          <StatCard
   474→            label="Wave Height"
   475→            value={fmtNum(data.waveHeight, 1)}
   476→            unit="m"
   477→            icon={Waves}
   478→            accent="blue"
   479→            sub={fmtNum(data.wavePeriod, 0) ? `${fmtNum(data.wavePeriod, 0)}s period` : undefined}
   480→          />
   481→          <StatCard
   482→            label="Air Temp"
   483→            value={fmtNum(data.airTemperature, 0)}
   484→            unit="°C"
   485→            icon={Thermometer}
   486→            accent="amber"
   487→          />
   488→          <StatCard
   489→            label="Sea Surface"
   490→            value={fmtNum(data.seaSurfaceTemperature, 0)}
   491→            unit="°C"
   492→            icon={Waves}
   493→            accent="green"
   494→          />
   495→        </div>
   496→      )}
   497→
   498→      {/* ── Cards grid ─────────────────────────────────────────────── */}
   499→      <section className="mt-4" aria-label="Sea condition details">
   500→        {loading && !data ? (
   501→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   502→            {Array.from({ length: 6 }).map((_, i) => (
   503→              <CardSkeleton key={i} />
   504→            ))}
   505→          </div>
   506→        ) : error ? (
   507→          <ErrorState
   508→            title="Couldn't load sea conditions"
   509→            message={error}
   510→            onRetry={() => coords && void fetchData(coords.lat, coords.lng)}
   511→          />
   512→        ) : data ? (
   513→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   514→            <LocationCard
   515→              label={label}
   516→              lat={coords?.lat ?? data.location?.lat ?? null}
   517→              lng={coords?.lng ?? data.location?.lng ?? null}
   518→              accuracy={accuracy}
   519→              onRefresh={() => coords && void fetchData(coords.lat, coords.lng)}
   520→              onUseMyLocation={useMyLocation}
   521→              loading={loading}
   522→            />
   523→            <WeatherOverviewCard data={data} />
   524→            <WindCard data={data} />
   525→            <WaveCard data={data} />
   526→            <TemperatureCard data={data} />
   527→            <AtmosphericsCard data={data} />
   528→            <SunCard data={data} />
   529→          </div>
   530→        ) : null}
   531→      </section>
   532→
   533→      {/* ── Marine condition chart ──────────────────────────────────── */}
   534→      <section className="mt-4" aria-label="24-hour marine forecast chart">
   535→        <MarineConditionChart data={chartData} loading={loading} />

   537→
   538→      {/* ── Footer ─────────────────────────────────────────────────── */}
   539→      <footer className="mt-6 flex flex-col gap-2 border-t border-border/60 pt-4 text-xs text-muted-foreground sm:flex-row sm:items-center sm:justify-between">
   540→        <div>
   541→          Last updated:{" "}
   542→          <span className="font-medium text-foreground">
   543→            {formatUpdatedAt(recordedAt)}
   544→          </span>
   545→          {source && <span className="ml-2">· source: {source}</span>}
   546→        </div>
   547→        {isDemo && (
   548→          <div className="flex items-center gap-1.5 text-amber-500">
   549→            <Info className="size-3.5" />
   550→            <span>
   551→              Live API unavailable — showing offline demo data.
   552→            </span>
   553→          </div>
   554→        )}
   555→      </footer>
   556→    </div>
   557→  );
   558→}
